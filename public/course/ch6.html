第17课：实战一——新闻网站爬取与多页采集

17.1 项目需求分析

目标：抓取某新闻网站列表页的新闻标题、链接、发布时间，并自动翻页抓取多页。

开发前必做的三件事（真实项目的标准流程）：

  1. 打开目标网站，按 F12 观察：请求是 GET 还是 POST？参数怎么翻页？
  2. 右键一条新闻标题 → 检查，确定标题、时间对应的 HTML 结构和 class 名；
  3. 检查 robots 协议（网址后加 /robots.txt）与网站使用条款，确认允许抓取，并规划好请求间隔。

17.2 完整代码

"""
项目一：新闻网站多页爬取
功能：抓取新闻标题、链接、发布时间，支持多页翻页
"""
import csv
import time
import requests
from bs4 import BeautifulSoup
from urllib import parse

HEADERS = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) '
                  'AppleWebKit/537.36 (KHTML, like Gecko) '
                  'Chrome/120.0.0.0 Safari/537.36'
}

def fetch_page(url):
    """下载一个页面，返回 BeautifulSoup 对象"""
    try:
        response = requests.get(url, headers=HEADERS, timeout=10)
        response.raise_for_status()
        response.encoding = response.apparent_encoding   # 防中文乱码
        return BeautifulSoup(response.text, 'lxml')
    except requests.exceptions.RequestException as e:
        print(f'  页面抓取失败 {url}: {e}')
        return None

def parse_news(soup, base_url):
    """解析一个页面中的所有新闻"""
    results = []
    # ⚠️ 以下选择器是示例，实战中要用 F12 确认目标网站的真实结构
    for item in soup.select('.news-item'):
        title_tag = item.select_one('a.title')
        time_tag = item.select_one('span.time')
        if title_tag:
            results.append({
                'title': title_tag.get_text(strip=True),
                'link': parse.urljoin(base_url, title_tag.get('href', '')),
                'time': time_tag.get_text(strip=True) if time_tag else ''
            })
    return results

def crawl_news(base_url, max_page=3):
    """主流程：循环翻页抓取"""
    all_news = []
    for page in range(1, max_page + 1):
        # 假设翻页规则是 ?page=N，实战中按 F12 观察到的真实规则修改
        url = f'{base_url}?page={page}'
        print(f'正在抓取第 {page} 页: {url}')

        soup = fetch_page(url)
        if soup is None:
            continue

        news = parse_news(soup, url)
        print(f'  本页抓到 {len(news)} 条新闻')
        all_news.extend(news)

        # 如果一页都抓不到，可能翻页到底了，提前结束
        if len(news) == 0:
            print('  没有更多内容，停止翻页')
            break

        time.sleep(1.5)   # 🕷️ 爬虫礼仪：每次请求间隔1.5秒
    return all_news

def save_to_csv(data, filename='新闻数据.csv'):
    """保存为 CSV（utf-8-sig 保证 Excel 打开不乱码）"""
    if not data:
        print('没有数据可保存')
        return
    with open(filename, 'w', encoding='utf-8-sig', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=['title', 'link', 'time'])
        writer.writeheader()
        writer.writerows(data)
    print(f'已保存 {len(data)} 条数据到 {filename}')

if __name__ == '__main__':
    BASE_URL = 'https://example.com/news'   # 换成你的目标网站
    news_list = crawl_news(BASE_URL, max_page=3)
    save_to_csv(news_list)

    # 打印前5条预览
    for i, n in enumerate(news_list[:5], 1):
        print(f"{i}. [{n['time']}] {n['title']}")

17.3 代码结构解读

这个项目的函数划分是爬虫项目的标准套路，以后写任何爬虫都可以套这个骨架：

fetch_page()   负责「下载」——只管把网页拿回来
parse_news()   负责「解析」——只管从 HTML 提取数据
crawl_news()   负责「调度」——翻页循环、间隔控制、终止判断
save_to_csv()  负责「存储」——数据落地

每个函数只干一件事，哪个环节出问题就调哪个，排查效率极高。

📝 本课练习

  1. 把项目改造成抓取「商品列表」（字段：商品名、价格、链接），并保存为 CSV。
  2. 增加断点续传功能：程序中断后重新运行时，跳过已抓取的页面（提示：用文件记录已完成的页码）。
  3. 增加去重逻辑：同一条新闻（按链接判断）不重复保存。

第18课：实战二——批量图片下载器

18.1 项目需求分析

目标：输入一个网页地址，自动找出页面中所有图片并批量下载到本地文件夹，按序号命名。

技术要点：

  • 提取图片地址：BeautifulSoup 找 <img> 标签 + urljoin 补全相对路径
  • 下载图片：必须用 response.content（二进制），不能用 .text
  • 文件名处理：从 URL 中提取后缀名，防止重名覆盖
  • 失败容忍：某张图下载失败不能影响其他图片

18.2 完整代码

"""
项目二：批量图片下载器
功能：抓取指定网页中的所有图片，保存到本地文件夹
"""
import os
import time
import requests
from bs4 import BeautifulSoup
from urllib import parse

HEADERS = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) '
                  'AppleWebKit/537.36 (KHTML, like Gecko) '
                  'Chrome/120.0.0.0 Safari/537.36'
}

def get_image_urls(page_url):
    """第一步：获取页面中所有图片的完整URL"""
    try:
        response = requests.get(page_url, headers=HEADERS, timeout=10)
        response.raise_for_status()
    except requests.exceptions.RequestException as e:
        print(f'页面获取失败: {e}')
        return []

    soup = BeautifulSoup(response.text, 'lxml')
    urls = []
    for img in soup.find_all('img'):
        src = img.get('src') or img.get('data-src')   # 兼容懒加载图片
        if src and not src.startswith('data:'):        # 排除base64内嵌图
            full_url = parse.urljoin(page_url, src)
            if full_url not in urls:
                urls.append(full_url)
    return urls

def download_image(img_url, save_dir, index):
    """第二步：下载单张图片"""
    # 从URL推断文件后缀，默认jpg
    path = parse.urlparse(img_url).path
    ext = os.path.splitext(path)[1].lower()
    if ext not in ['.jpg', '.jpeg', '.png', '.gif', '.webp']:
        ext = '.jpg'

    filename = os.path.join(save_dir, f'img_{index:03d}{ext}')

    # 图片防盗链常见对策：Referer 声明来源页面
    headers = {**HEADERS, 'Referer': img_url}

    try:
        response = requests.get(img_url, headers=headers, timeout=15)
        response.raise_for_status()
        with open(filename, 'wb') as f:        # 'wb' 二进制写模式
            f.write(response.content)
        return True
    except requests.exceptions.RequestException as e:
        print(f'  下载失败 {img_url}: {e}')
        return False

def download_all(page_url, save_dir='下载的图片'):
    """主流程"""
    # 创建保存目录（exist_ok=True 避免目录已存在时报错）
    os.makedirs(save_dir, exist_ok=True)

    print(f'正在分析页面: {page_url}')
    img_urls = get_image_urls(page_url)
    print(f'共发现 {len(img_urls)} 张图片，开始下载...\n')

    success = 0
    for i, url in enumerate(img_urls, 1):
        if download_image(url, save_dir, i):
            success += 1
            print(f'  [{i}/{len(img_urls)}] 成功: {url}')
        time.sleep(0.5)   # 下载间隔，礼貌爬取

    print(f'\n完成！成功 {success} 张，失败 {len(img_urls) - success} 张')
    print(f'保存位置: {os.path.abspath(save_dir)}')

if __name__ == '__main__':
    download_all('https://example.com')   # 换成你的目标网页

18.3 关键知识点回顾

知识点 出处 本项目中的应用
response.content 二进制 第二章 图片必须用二进制写入
Referer 防盗链 第二章 部分网站图片直接访问会 403
urljoin 补全相对路径 第三章 网页里图片地址常是 /img/xx.jpg
data-src 懒加载 第四章 很多网站图片真实地址藏在 data-src
os.makedirs 文件操作 自动创建保存目录

📝 本课练习

  1. 给下载器增加「文件大小过滤」：小于 10KB 的图片（通常是图标）不下载（提示：看 len(response.content)）。
  2. 增加「按图片在网页中的顺序，用网页标题命名文件」的功能。
  3. 思考：如果网站图片是滚动加载（翻到底才显示），requests 方案为什么抓不全？（为进阶课 Selenium 埋下伏笔）

目标：把前面抓到的新闻数据做进一步处理——存入文件，并做一个简单的词频统计，回答「最近新闻里什么词最热」。

方式 适合场景 难度

19.2 完整代码

本项目使用 jieba 中文分词库把标题切成词语，先安装：pip install jieba （第五章的正则也能切词，但中文没有空格分隔，专业分词库效果要好得多）

"""
项目三：新闻数据存储与热词分析
流程：抓取 → 存储CSV → 读取 → 词频统计 → 输出报告
"""
import csv
import jieba
from collections import Counter

# ===== 第一部分：数据存储（复用项目一的核心逻辑） =====

def save_to_csv(data, filename):
    with open(filename, 'w', encoding='utf-8-sig', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=['title', 'link', 'time'])
        writer.writeheader()
        writer.writerows(data)

def load_from_csv(filename):
    """从CSV读取数据"""
    with open(filename, 'r', encoding='utf-8-sig') as f:
        return list(csv.DictReader(f))

# ===== 第二部分：数据分析 =====

# 停用词：出现频率高但没有分析价值的词
STOP_WORDS = {'的', '了', '在', '是', '和', '与', '为', '对', '将',
              '年', '月', '日', '一个', '我们', '你们', '他们', 'the', 'a', 'to'}

def analyze_hot_words(news_list, top_n=20):
    """统计新闻标题中的高频词"""
    all_words = []
    for news in news_list:
        # jieba.lcut 把标题切成词语列表
        words = jieba.lcut(news['title'])
        # 过滤：去掉单字词、空白和停用词
        words = [w.strip() for w in words
                 if len(w.strip()) >= 2 and w.strip() not in STOP_WORDS]
        all_words.extend(words)

    # Counter 一行完成词频统计
    counter = Counter(all_words)
    return counter.most_common(top_n)

def generate_report(news_list, hot_words, filename='分析报告.txt'):
    """生成文字分析报告"""
    lines = []
    lines.append('=' * 40)
    lines.append('新闻数据热词分析报告')
    lines.append('=' * 40)
    lines.append(f'数据总量: {len(news_list)} 条新闻')
    lines.append('')
    lines.append(f'【TOP {len(hot_words)} 热词榜】')
    for rank, (word, count) in enumerate(hot_words, 1):
        bar = '█' * count   # 用文字画个简单的柱状图
        lines.append(f'{rank:>2}. {word:<8} {count:>3}次 {bar}')

    report = '\n'.join(lines)
    with open(filename, 'w', encoding='utf-8') as f:
        f.write(report)
    print(report)
    print(f'\n报告已保存到 {filename}')

if __name__ == '__main__':
    # 模拟已抓取的数据（实际使用时替换为项目一 crawl_news() 的结果）
    demo_data = [
        {'title': '人工智能助力医疗行业数字化转型', 'link': 'http://a.com/1', 'time': '2024-01-15'},
        {'title': '新能源汽车销量创新高 行业前景广阔', 'link': 'http://a.com/2', 'time': '2024-01-15'},
        {'title': '人工智能技术在教育领域的应用探索', 'link': 'http://a.com/3', 'time': '2024-01-16'},
        {'title': '医疗行业迎来新一轮数字化升级浪潮', 'link': 'http://a.com/4', 'time': '2024-01-16'},
        {'title': '新能源电池技术取得重大突破', 'link': 'http://a.com/5', 'time': '2024-01-17'},
    ]

    # 1. 存储
    save_to_csv(demo_data, '新闻数据.csv')

    # 2. 读取
    news_list = load_from_csv('新闻数据.csv')

    # 3. 分析
    hot_words = analyze_hot_words(news_list, top_n=10)

    # 4. 报告
    generate_report(news_list, hot_words)
